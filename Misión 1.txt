def calcular_envio(total, region):
    """Devuelve el costo de envío en pesos chilenos.

    Reglas acordadas con el área comercial:
      - Compras de $50.000 o más: envío gratis ($0), a cualquier región.
      - Compras menores a $50.000 en la Región Metropolitana ("RM"): $3.990.
      - Compras menores a $50.000 en cualquier otra región: $6.990.
      - La región puede venir escrita en mayúsculas o minúsculas.
    """
    if total >= 50000:
        return 0
    if region.upper() == "RM":
        return 3990
    return 6990

def descuento_por_cantidad(cantidad):
    """Devuelve el porcentaje de descuento (0, 10 o 20) según la cantidad comprada."""
    if cantidad <= 0:
        raise ValueError("La cantidad debe ser mayor que cero")
    if cantidad < 5:
        return 0
    elif cantidad < 10:
        return 10
    else:
        return 20
